Lua គឺជាភាសាទីពីរនៃ[ផ្ទាំងការងារស្គ្រីប](topic:craftscript)។ ជ្រើសរើស **Lua** នៅក្នុងប្រអប់ភាសាពីលើកម្មវិធីកែ (ឯកសារ `.lua` បើកជា Lua ដោយខ្លួនឯង)។ Lua និង CraftScript ចែករំលែកអថេរដូចគ្នា និងអនុគមន៍ដែលមានស្រាប់ដូចគ្នា ដូច្នេះស្គ្រីប Lua អាចប្រើអ្វីដែលស្គ្រីប CraftScript បានបង្កើត ហើយផ្ទុយមកវិញ។

## ចំណុចសំខាន់

```
-- អនុគមន៍ដែលមានស្រាប់ទាំងអស់នៅក្នុងតារាង cs
local K = cs.zeros(3, 3)
K:set(1, 1, 4)                 -- ចាប់ពី 1 ដូច CraftScript
print(K(1, 1), #K)             -- ធាតុមួយ; ចំនួនធាតុ

-- ផ្ទាំងការងារ៖ អថេរដូចគ្នាដែលស្គ្រីបគ្រប់ដួងមើលឃើញ
ws.total = K(1, 1) * 2
print(ws.total)

-- លទ្ធផលជាច្រើនពីអនុគមន៍មួយ
local V, D = cs.call('eig', 2, cs.tensor({{2, 1}, {1, 2}}))
```

- `cs.<name>(…)` ហៅអនុគមន៍ដែលមានស្រាប់ណាមួយ ហើយត្រឡប់លទ្ធផលទីមួយរបស់វា; `cs.call(name, n, …)` ត្រឡប់ `n` លទ្ធផល។
- `cs.tensor({1, 2, 3})` បង្កើតជួរដេកមួយ `cs.tensor({{1, 2}, {3, 4}})` បង្កើតម៉ាទ្រីសមួយ។
- អារ៉ាយ ម៉ាទ្រីសស្ពារ ម៉ែស និងវាលគឺជាតម្លៃដែលមានវិធីសាស្ត្រ៖ `t(i, j)` ឬ `t:get(i, j)`, `t:set(i, j, v)`, `t:shape()`, `#t`, `t:T()` (ផ្លាស់ប្តូរទិស), `t:totable()`។ `+ - * / ^` មានន័យដូចក្នុង CraftScript (`*` គឺជាផលគុណម៉ាទ្រីស); `t:times(u)` គុណធាតុម្តងមួយ។
- លេខដែលជាអារ៉ាយ 1×1 មកដល់ជាលេខ Lua ធម្មតា។
- `ws.name` អាននិងសរសេរផ្ទាំងការងារដែលចែករំលែក។ អថេរសកលដែលស្គ្រីបរបស់អ្នកទុកចោលក៏ចូលទៅទីនោះនៅពេលវាបញ្ចប់ដែរ។
- `print(…)` សរសេរទៅកុងសូល; `cs.show(f)`, `cs.plot(y)` និងអនុគមន៍បង្ហាញផ្សេងទៀតដំណើរការដូចក្នុង CraftScript។

## ហៅបណ្ណាល័យវិភាគ

អនុគមន៍នៃបណ្ណាល័យវិភាគ (ផ្ទាំង បណ្ណាល័យ) និងអនុគមន៍ CraftScript ដែលស្គ្រីបរបស់គ្រឿងអ្នកកំណត់ ក៏ត្រូវបានទៅដល់តាម `cs` ដែរ៖

```lua
local t = cs.box_mesh(100, 10, 10, 5, 'tet10')
local bc = cs.struct({ fixedFaces = 5, loads = { cs.struct({ face = 6, force = cs.tensor({ 0, -100, 0 }) }) } })
local r = cs.static_tet(t, 200000, 0.3, bc)
print(r.maxDisp)
local nodes, tris = cs.call('face_nodes', 2, t, 6)   -- លទ្ធផលពីរ
```

វាត្រូវបានរកឃើញមុនពេលស្គ្រីបចាប់ផ្តើម ដូច្នេះសូមសរសេរឈ្មោះក្នុងស្គ្រីបដោយផ្ទាល់៖ `cs.static_tet(` ឬ `cs.call('static_tet', …)`។ ឈ្មោះដែលត្រូវបានបង្កើតពេលស្គ្រីបកំពុងដំណើរការ គ្រាន់តែទៅដល់អនុគមន៍ដែលមានស្រាប់។ រចនាសម្ព័ន្ធដែលបង្កើតដោយ `cs.struct({ … })` គឺជារចនាសម្ព័ន្ធ CraftScript ហើយបញ្ជី Lua នៃពួកវា (`{ cs.struct(…), … }`) គឺជាក្រឡា ដែលជាអ្វីដែលការកំណត់របស់បណ្ណាល័យរំពឹង។ បើមានអ្វីខុសនៅក្នុងអនុគមន៍បណ្ណាល័យ សារនឹងដាក់ឈ្មោះអនុគមន៍ និងបន្ទាត់របស់វា ហើយចង្អុលទៅបន្ទាត់របស់អ្នកដែលបានហៅវា។

## ត្រូវបានបិទ

ស្គ្រីបមិនអាចទៅដល់ឯកសារ បណ្តាញ ឬទំព័របានទេ៖ `io`, `os` (លើកលែងតែនាឡិកា), `require`, `dofile`, `loadfile` និង `debug` ត្រូវបានដកចេញ។ ប្រើ Open របស់ផ្ទាំងការងារស្គ្រីប និងអថេរ ដើម្បីនាំទិន្នន័យចូល។ ថវិកាជំហាន និង Stop ដំណើរការដូចក្នុង CraftScript។
