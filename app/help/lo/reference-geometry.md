ເລຂາຄະນິດອ້າງອີງແມ່ນໂຄງຄ້ຳທີ່ທ່ານສ້າງຟີເຈີໃສ່: **ຍົນ** ເພື່ອວາງພາບຮ່າງບ່ອນທີ່ບໍ່ມີໜ້າ, **ແກນ** ເພື່ອໝຸນ ຫຼືເຮັດແບບຮອບມັນ, **ຈຸດ** ເພື່ອຢຸດການ Extrude. ມັນບໍ່ມີເນື້ອວັດຖຸຂອງຕົວເອງ. ຍົນຈະສະແດງໃນມຸມມອງເປັນສີ່ຫຼ່ຽມສີອ່ອນ, ແກນເປັນເສັ້ນປະ, ຈຸດເປັນເຄື່ອງໝາຍກາກບາດນ້ອຍ; ທັງສາມຈະເຂົ້າໄປໃນປະຫວັດ ແລະຕິດຕາມສິ່ງທີ່ມັນຖືກສ້າງຈາກ ເມື່ອທ່ານປ່ຽນຂະໜາດທີ່ຢູ່ເບື້ອງເທິງ.

## ສ້າງຍົນ

1. ຄລິກໜ້າແບນ, ຫຼືຍົນຖານໃນປະຫວັດ ເພື່ອເລີ່ມຈາກມັນ (ຕົວເລືອກ).
2. ເລືອກ {{ui:More}} ແລ້ວ {{ui:Plane}}. ມັນເລີ່ມເປັນຍົນຊົດເຊີຍ 20 % ຂອງຂະໜາດຊິ້ນສ່ວນ ຫ່າງຈາກສິ່ງທີ່ທ່ານເລືອກ (ຫຼືຈາກຍົນເທິງສຸດ).
3. ໃນແຜງຂວາ ເລືອກ **ປະເພດ** ແລະການອ້າງອີງຂອງມັນ:

| ປະເພດ | ການອ້າງອີງ | ຕົວເລກ |
|---|---|---|
| ຊົດເຊີຍຈາກຍົນ ຫຼືໜ້າ | **ຈາກຍົນ** | **ໄລຍະ (mm)**, ລົບສຳລັບອີກດ້ານ |
| ເປັນມຸມ, ຜ່ານຂອບ ຫຼືແກນ | **ຈາກຍົນ**, **ໝຸນຮອບ** (ແກນໂລກ, ຟີເຈີແກນ, ຫຼືຂອບຊື່ທີ່ເລືອກ) | **ມຸມ (°)** |
| ຜ່ານສາມຈຸດ | **ຈຸດ 1**, **ຈຸດ 2**, **ຈຸດ 3** | — |
| ຍົນກາງລະຫວ່າງສອງ | **ຍົນທຳອິດ**, **ຍົນທີສອງ** (ຂະໜານ: ເຄິ່ງທາງ; ມາພົບກັນ: ຍົນທີ່ແບ່ງມຸມເຄິ່ງໜຶ່ງ) | — |
| ສຳຜັດກັບຊິລິນເດີ | **ຊິລິນເດີ** (ໜ້າຊິລິນເດີ), **ຈຸດໃກ້ຄຽງ** (ເລືອກດ້ານ) | — |
| ຕັ້ງສາກກັບຂອບ ທີ່ຈຸດ | **ຂອບ**, **ທີ່ຈຸດ** (ຍົນຜ່ານຈຸດຂອງຂອບທີ່ໃກ້ທີ່ສຸດ ແລະຜ່ານຈຸດນີ້ດ້ວຍ) | — |
| ຜ່ານຂອບ ແລະຈຸດ | **ຂອບຊື່** (ຫຼືແກນ), **ຜ່ານຈຸດ** | — |
| ຕັ້ງສາກກັບຂອບຢູ່ກາງ | **ຂອບ** (ເຄິ່ງທາງຕາມມັນ, ຕັ້ງສາກກັບມັນຕຽງນັ້ນ), ຫຼືບໍ່ມີຂອບ **ຈຸດ 1** ແລະ **ຈຸດ 2** (ຍົນເຄິ່ງທາງລະຫວ່າງສອງຈຸດ) | — |

ທຸກການອ້າງອີງມີປຸ່ມ **ເລືອກ**: ເລືອກມັນ ແລ້ວຄລິກໃນມຸມມອງ. ຊ່ອງຍົນຮັບຍົນຖານ, ຍົນອ້າງອີງ ຫຼືໜ້າຮາບ; ຊ່ອງຈຸດຮັບປາຍຂອງຂອບ (ຄລິກໃກ້ປາຍນັ້ນ), ຟີເຈີຈຸດ, ຫຼືຈຸດໜຶ່ງໃນໜ້າ; ຊ່ອງຂອບຮັບຂອບ. {{key:Esc}} ຍົກເລີກການເລືອກ. **ກັບດ້ານເສັ້ນປົກກະຕິ** ຫັນຍົນກັບ (ພາບຮ່າງທີ່ຢູ່ເທິງມັນຫັນໄປທາງໃດ).

ໃນເວີຊັນນີ້, ຊ່ອງທີ່ຕ້ອງການຂອບ ຫຼືໜ້າຊິລິນເດີ (ສຳຜັດກັບຊິລິນເດີ ແລະຊ່ອງຂອບ) ຈະຕອບວ່າ "ຍັງບໍ່ໄດ້ສ້າງ": ໃຫ້ໃຊ້ຈຸດ ຫຼືແກນແທນ.

## ວາງພາບຮ່າງເທິງຍົນ

ເລືອກຍົນ (ໃນປະຫວັດ ຫຼືຄລິກສີ່ຫຼ່ຽມຂອງມັນໃນມຸມມອງ) ແລ້ວເລືອກ {{ui:Sketch}}, ຫຼື **ວາງພາບຮ່າງເທິງມັນ** ໃນແຜງຂອງມັນ. ພາບຮ່າງຈະຕາມຍົນ: ຖ້າທ່ານປ່ຽນໄລຍະຂອງຍົນ, ພາບຮ່າງ ແລະທຸກຟີເຈີເທິງມັນກໍຈະຍ້າຍຕາມ.

## ແກນ ແລະຈຸດ

- {{ui:Axis}}: ຜ່ານສອງຈຸດ, ແກນຂອງຊິລິນເດີ ຫຼືໂກນ, ບ່ອນທີ່ສອງຍົນມາພົບກັນ, ຕາມຂອບຊື່ (ຫຼືແກນຂອງຂອບວົງມົນ), ຫຼືຕັ້ງສາກກັບໜ້າຜ່ານຈຸດໜຶ່ງ.
- {{ui:Point}}: ຢູ່ພິກັດ, ທີ່ຈຸດຍອດ ຫຼືຈຸດພາບຮ່າງ, ຫຼືຢູ່ຈຸດກາງຂອງຂອບວົງມົນ, ພ້ອມຈຸດ "ລະຫວ່າງວັດຖຸ":

| ປະເພດຈຸດ | ການອ້າງອີງ |
|---|---|
| ຈຸດກາງຂອງຂອບ (ຫຼືລະຫວ່າງສອງຈຸດ) | **ຂອບ** (ເຄິ່ງທາງຕາມຄວາມຍາວ, ລວມທັງເສັ້ນໂຄ້ງ), ຫຼືບໍ່ມີຂອບ **ຈຸດ 1** ແລະ **ຈຸດ 2** |
| ບ່ອນທີ່ຂອບ ຫຼືແກນມາພົບໜ້າ ຫຼືຍົນ | **ຂອບ** (ຊື່ ຫຼືໂຄ້ງ) ຫຼືແກນ, ແລະ **ພົບຍົນ** (ຍົນ ຫຼືໜ້າຮາບ) ຫຼື **ຫຼືໜ້າ** (ໜ້າຮາບ ຫຼືຊິລິນເດີ). ວົງມົນຕັດຍົນສອງຄັ້ງ ແລະເສັ້ນຜ່ານຊິລິນເດີສອງຄັ້ງ: **ການຕັດອີກຄັ້ງ** ຈະເອົາຄັ້ງທີສອງ |
| ບ່ອນທີ່ສາມຍົນມາພົບກັນ | **ຍົນທຳອິດ**, **ຍົນທີສອງ** ແລະ **ຍົນທີສາມ** (ຍົນ ຫຼືໜ້າຮາບ) |

ແຕ່ລະອັນຈະລົ້ມເຫຼວດ້ວຍຂໍ້ຄວາມທີ່ລຽບງ່າຍ ເມື່ອການອ້າງອີງບໍ່ເໝາະ: ເສັ້ນຂະໜານກັບຍົນ, ເສັ້ນທີ່ພາດຊິລິນເດີ, ຂອບທີ່ບໍ່ຮອດຍົນ, ສາມຍົນທີ່ບໍ່ພົບກັນໃນຈຸດດຽວ, ຫຼືຈຸດທີ່ຢູ່ເທິງເສັ້ນທີ່ມັນຄວນກຳນົດຍົນ.

ໃຊ້ມັນໃນຟີເຈີອື່ນ:

| ບ່ອນ | ວິທີ |
|---|---|
| Revolve | **Axis**: ເລືອກຟີເຈີແກນຈາກລາຍການ (ມັນຕ້ອງຢູ່ໃນຍົນຂອງພາບຮ່າງ) |
| Circular pattern | **Axis**: ຟີເຈີແກນ |
| Linear pattern | **Direction**: ຟີເຈີແກນ, ຫຼືຍົນ (ເສັ້ນປົກກະຕິຂອງມັນ) |
| Mirror | **Mirror plane**: ຍົນອ້າງອີງ |
| Extrude | **End**: ຮອດໜ້າ ຫຼືຍົນ, ຮອດຈຸດຍອດ (ຟີເຈີຈຸດ). ຍັງບໍ່ໄດ້ສ້າງໃນເວີຊັນນີ້ (ມັນຕ້ອງການ exact kernel) |
| Draft | **ຍົນເປັນກາງ** |

## ເມື່ອການອ້າງອີງຫາຍໄປ

ຍົນທີ່ສ້າງຈາກໜ້າທີ່ບໍ່ມີແລ້ວຈະລົ້ມເຫຼວ ພ້ອມຂໍ້ຄວາມທີ່ລະບຸມັນ ເຊັ່ນ "ໜ້າທີ່ມັນຕັ້ງຢູ່ຫາຍໄປຈາກ Extrude1." ມັນຈະບໍ່ຖືກຍ້າຍໄປໜ້າອື່ນເດັດຂາດ. ພາບຮ່າງ ແລະຟີເຈີເທິງມັນຈະຖືກໝາຍວ່າ "ຍັງບໍ່ໄດ້ສ້າງ" ຈົນກວ່າທ່ານຈະເລືອກການອ້າງອີງໃໝ່; ສ່ວນທີ່ເຫຼືອຂອງຊິ້ນສ່ວນຈະຢູ່ຄືເກົ່າ.

:::note
ເວີຊັນນີ້ສ້າງທຸກຊິ້ນສ່ວນດ້ວຍ mesh kernel, ຊຶ່ງສ້າງຍົນ, ແກນ ແລະຈຸດຈາກຍົນຖານ, ໜ້າຮາບທີ່ສ້າງດ້ວຍ Extrude ຫຼື Revolve, ຈຸດພາບຮ່າງ, ພິກັດ ແລະຟີເຈີອ້າງອີງອື່ນໆ. ສິ່ງໃດກໍຕາມທີ່ຕ້ອງການຂອບ, ປາຍຂອງຂອບ ຫຼືໜ້າໂຄ້ງ ຕ້ອງໃຊ້ exact kernel: ຍັງບໍ່ໄດ້ສ້າງໃນເວີຊັນນີ້ ("ມັນໃຊ້ໜ້າ ຫຼືຂອບຂອງຮ່າງທຶບ, ຊຶ່ງມີແຕ່ exact kernel ເທົ່ານັ້ນທີ່ຊອກພົບໄດ້: ຍັງບໍ່ໄດ້ສ້າງໃນເວີຊັນນີ້.").
:::
